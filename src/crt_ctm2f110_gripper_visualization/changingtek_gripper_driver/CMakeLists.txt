cmake_minimum_required(VERSION 3.16)
project(changingtek_gripper_driver)

find_package(ament_cmake REQUIRED)
find_package(control_msgs REQUIRED)
find_package(rclcpp REQUIRED)
find_package(rclcpp_action REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(trajectory_msgs REQUIRED)

set(SDK_DIR "${CMAKE_CURRENT_SOURCE_DIR}/vendor/ctm2f110_sdk")
if(CMAKE_SYSTEM_PROCESSOR MATCHES "^(aarch64|arm64)")
  set(SDK_LIB_DIR "${SDK_DIR}/include/arm64")
  set(SDK_INSTALL_LIB_DIR "arm64")
else()
  set(SDK_LIB_DIR "${SDK_DIR}/include/x86_64")
  set(SDK_INSTALL_LIB_DIR "x86_64")
endif()

add_executable(ctm2f110_driver src/ctm2f110_driver.cpp "${SDK_DIR}/gripper.cpp")
target_include_directories(ctm2f110_driver PRIVATE "${SDK_DIR}" "${SDK_DIR}/include")
target_link_directories(ctm2f110_driver PRIVATE "${SDK_LIB_DIR}")
target_link_libraries(ctm2f110_driver CRTServo)
target_compile_features(ctm2f110_driver PRIVATE cxx_std_17)
target_compile_options(ctm2f110_driver PRIVATE -Wall -Wextra -Wpedantic)
set_target_properties(ctm2f110_driver PROPERTIES
  BUILD_RPATH "${SDK_LIB_DIR}"
  INSTALL_RPATH "\$ORIGIN/../../share/${PROJECT_NAME}/vendor/ctm2f110_sdk/include/${SDK_INSTALL_LIB_DIR}")
ament_target_dependencies(ctm2f110_driver control_msgs rclcpp rclcpp_action sensor_msgs trajectory_msgs)
install(TARGETS ctm2f110_driver DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY launch config DESTINATION share/${PROJECT_NAME})
install(DIRECTORY vendor/ctm2f110_sdk DESTINATION share/${PROJECT_NAME}/vendor)
install(FILES README.md DESTINATION share/${PROJECT_NAME})
ament_package()
