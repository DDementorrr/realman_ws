cmake_minimum_required(VERSION 3.16)
project(rm65b_operator_panel)
find_package(ament_cmake REQUIRED)
find_package(tl_expected REQUIRED)
find_package(tl-expected CONFIG REQUIRED)
find_package(Qt5 REQUIRED COMPONENTS Widgets)
foreach(dep rviz_common rviz_rendering moveit_ros_visualization pluginlib rclcpp geometry_msgs sensor_msgs std_msgs std_srvs tf2_ros tf2_eigen visualization_msgs)
  find_package(${dep} REQUIRED)
endforeach()
set(CMAKE_AUTOMOC ON)
add_library(${PROJECT_NAME} SHARED src/operator_panel.cpp include/rm65b_operator_panel/operator_panel.hpp)
target_include_directories(${PROJECT_NAME} PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
target_compile_features(${PROJECT_NAME} PUBLIC cxx_std_17)
target_compile_options(${PROJECT_NAME} PRIVATE -Wall -Wextra -Wpedantic)
ament_target_dependencies(${PROJECT_NAME} rviz_common moveit_ros_visualization pluginlib rclcpp geometry_msgs sensor_msgs std_msgs std_srvs tf2_ros tf2_eigen visualization_msgs)
target_link_libraries(${PROJECT_NAME} Qt5::Widgets moveit_ros_visualization::moveit_motion_planning_rviz_plugin_core)
pluginlib_export_plugin_description_file(rviz_common plugin_description.xml)
install(TARGETS ${PROJECT_NAME} LIBRARY DESTINATION lib)
install(DIRECTORY include/ DESTINATION include)

if(BUILD_TESTING)
  find_package(Qt5 REQUIRED COMPONENTS Test)
  add_executable(native_panel_probe test/native_panel_probe.cpp)
  target_compile_features(native_panel_probe PUBLIC cxx_std_17)
  ament_target_dependencies(native_panel_probe rviz_common rviz_rendering moveit_ros_visualization rclcpp geometry_msgs sensor_msgs std_msgs visualization_msgs)
  target_link_libraries(native_panel_probe Qt5::Widgets Qt5::Test moveit_ros_visualization::moveit_motion_planning_rviz_plugin_core)
  install(TARGETS native_panel_probe DESTINATION lib/${PROJECT_NAME})
endif()

ament_package()
